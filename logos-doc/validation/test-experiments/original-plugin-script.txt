package testplugin

import (
	"os/exec"
	"path/filepath"
	"plugin"
	"testing"
)

func TestChildPlugin(t *testing.T) {
	path := filepath.Join(t.TempDir(), "child.so")
	out, err := exec.Command("go", "build", "-buildmode=plugin", "-o", path, "./plugin").CombinedOutput()
	if err != nil {
		t.Fatalf("build child plugin: %v\n%s", err, out)
	}
	loaded, err := plugin.Open(path)
	if err != nil {
		t.Fatal(err)
	}
	symbol, err := loaded.Lookup("Answer")
	if err != nil {
		t.Fatal(err)
	}
	answer, ok := symbol.(func() int)
	if !ok {
		t.Fatalf("Answer has type %T, want func() int", symbol)
	}
	if got := answer(); got != 42 {
		t.Fatalf("plugin Answer = %d, want 42", got)
	}
}
